async function uploadFilm() { const f = document.getElementById('videoFile').files[0]; const filmTitle = document.getElementById('title').value.trim(); const filmYear = document.getElementById('year').value.trim(); const filmGenre = document.getElementById('genre').value.trim(); const pf = document.getElementById('posterFile').files[0]; const statusEl = document.getElementById('status'); if (!filmTitle || !f) { statusEl.textContent = 'Judul dan video wajib diisi.'; return; } if (f.size > 190 * 1024 * 1024) { statusEl.textContent = 'Video maksimal 190 MB untuk metode upload ini.'; return; } if (pf && pf.size > 5 * 1024 * 1024) { statusEl.textContent = 'Poster maksimal 5 MB.'; return; } if (!session || !session.access_token) { statusEl.textContent = 'Sesi login habis. Silakan login kembali.'; return; } statusEl.textContent = 'Menyiapkan upload Cloudflare...'; let posterUrl = ''; let posterPath = ''; let uploadUid = ''; try { // Meminta tautan upload aman dari server. const response = await fetch('/api/stream-upload', { method: 'POST', headers: { Authorization: 'Bearer ' + session.access_token } }); const ticket = await response.json(); if (!response.ok || !ticket.uploadURL || !ticket.uid) { throw new Error( ticket.error || 'Gagal membuat tautan upload Cloudflare.' ); } uploadUid = ticket.uid; // Mengunggah video langsung ke Cloudflare Stream. statusEl.textContent = 'Mengupload video ke Cloudflare...'; const form = new FormData(); form.append('file', f); const uploaded = await fetch(ticket.uploadURL, { method: 'POST', body: form }); if (!uploaded.ok) { throw new Error( 'Upload video gagal. Periksa koneksi dan coba lagi.' ); } // Mengunggah poster ke Supabase jika dipilih. if (pf) { statusEl.textContent = 'Mengupload poster...'; const safe = pf.name.replace(/[^a-zA-Z0-9._-]/g, '_'); posterPath = crypto.randomUUID() + '-' + safe; const posterUpload = await db.storage .from('posters') .upload(posterPath, pf, { contentType: pf.type, upsert: false }); if (posterUpload.error) { throw new Error( 'Upload poster gagal: ' + posterUpload.error.message ); } posterUrl = db.storage .from('posters') .getPublicUrl(posterPath).data.publicUrl; } // Menyimpan informasi film ke database. statusEl.textContent = 'Menyimpan data film...'; const result = await db.from('films').insert({ title: filmTitle, year: filmYear, genre: filmGenre, poster_url: posterUrl, video_url: 'https://iframe.videodelivery.net/' + uploadUid, storage_path: 'cloudflare:' + uploadUid }); if (result.error) { throw new Error( 'Data film gagal disimpan: ' + result.error.message ); } statusEl.textContent = 'Film berhasil disimpan! Tunggu pemrosesan video selesai.'; document.getElementById('title').value = ''; document.getElementById('year').value = ''; document.getElementById('genre').value = ''; document.getElementById('posterFile').value = ''; document.getElementById('posterInfo').textContent = ''; document.getElementById('videoFile').value = ''; document.getElementById('fileInfo').textContent = ''; await loadFilms(); } catch (error) { if (posterPath) { await db.storage.from('posters').remove([posterPath]); } statusEl.textContent = error.message || 'Upload gagal.'; } }
